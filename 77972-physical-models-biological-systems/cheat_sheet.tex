\chapter{Model formulation and validation}\label{chap:cheatsheet-modelling}
\section{Modelling cycle}\label{sec:cheat-cycle}
\begin{enumerate}
    \item Formulate: specify question, variables, processes, assumptions, and observations.
    \item Solve: derive consequences analytically where possible; use computation for remaining behaviour.
    \item Validate: compare with observations, limiting cases, units, and invariants; revise assumptions when needed.
\end{enumerate}

\section{Inference from dynamics}\label{sec:cheat-inference}
For a model $\dot{x}=f(x,\theta)$, fitting estimates parameters $\theta$ from measured trajectories. Count independent features rather than raw data points; correlated measurements may contain less independent information. Structural identifiability asks whether distinct parameter values can produce exactly the same ideal observable data.

\chapter{Population turnover}\label{chap:cheatsheet-turnover}
\section{Input, output, and steady state}\label{sec:cheat-flux}
\begin{equation*}
    \frac{\dd W}{\dd t}=Q_{\mathrm{in}}-Q_{\mathrm{out}},
    \qquad \text{steady state: } Q_{\mathrm{in}}=Q_{\mathrm{out}}.
\end{equation*}
A steady amount does not determine the magnitude of the balanced fluxes. For constant input and first-order removal,
\begin{equation*}
    \frac{\dd W}{\dd t}=Q_{\mathrm{in}}-\kappa W,
    \qquad W^*=\frac{Q_{\mathrm{in}}}{\kappa},
    \qquad W(t)-W^*=[W(0)-W^*]\ee^{-\kappa t}.
\end{equation*}

\section{Two-stage viral clearance after blocking new infection}\label{sec:cheat-hiv}
With constant infected-cell clearance $k_I$, virion clearance $k_V$, production per infected cell $\gamma$, and $\beta=\gamma N_{I0}$:
\begin{equation*}
    \dot N_I=-k_I N_I,
    \qquad N_I(t)=N_{I0}\ee^{-k_It},
    \qquad \dot N_V=\beta\ee^{-k_It}-k_VN_V.
\end{equation*}
General solution, for $k_V\ne k_I$:
\begin{equation*}
    N_V(t)=N_{V0}\ee^{-k_Vt}
      +\frac{\beta}{k_V-k_I}
       (\ee^{-k_It}-\ee^{-k_Vt}).
\end{equation*}
If pretreatment viral production and clearance balance, $\beta=k_VN_{V0}$:
\begin{equation*}
    \frac{N_V(t)}{N_{V0}}
      =\frac{k_V\ee^{-k_It}-k_I\ee^{-k_Vt}}{k_V-k_I},
    \qquad \dot N_V(0)=0.
\end{equation*}
Equal-rate limit $k_I=k_V=k$:
\begin{equation*}
    N_V(t)=(N_{V0}+\beta t)\ee^{-kt},
    \qquad \beta=kN_{V0}\Longrightarrow
    N_V(t)=N_{V0}(1+kt)\ee^{-kt}.
\end{equation*}
This limit also has zero initial slope. Under the steady-state constraint, exchanging $k_I$ and $k_V$ leaves the normalized viral-load curve unchanged; the curve alone therefore cannot assign the two rate constants to their biological processes.

\section{First-order decay scales}\label{sec:cheat-decay}
For $N(t)=N_0\ee^{-kt}$, the half-life is $t_{1/2}=\ln 2/k$. A factor-$F$ decrease over interval $\Delta t$ corresponds to average apparent rate $r=\ln F/\Delta t$ if summarized by a single exponential; for a multiexponential trajectory this is not necessarily an individual process rate.
